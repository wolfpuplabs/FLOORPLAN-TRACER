// Konfigurasi ukuran const length = 5; // Panjang tembok const height = 2.5; // Tinggi tembok const thickness = 0.2; // Ketebalan tembok // 1. Gambar Tembok Utama (Shape 2D) const wallShape = new THREE.Shape(); wallShape.moveTo(0, 0); wallShape.lineTo(length, 0); wallShape.lineTo(length, height); wallShape.lineTo(0, height); wallShape.lineTo(0, 0); // Tutup jalur // 2. Gambar Lubang Pintu (Path 2D) // Misalnya pintu ada di jarak 2m dari kiri, lebar 1m, tinggi 2m const doorHole = new THREE.Path(); doorHole.moveTo(2, 0); doorHole.lineTo(3, 0); doorHole.lineTo(3, 2); doorHole.lineTo(2, 2); doorHole.lineTo(2, 0); // Tutup jalur lubang // 3. Masukkan lubang (Boolean subtraction otomatis) wallShape.holes.push(doorHole); // 4. Extrude (Tarik) Shape 2D tersebut menjadi 3D const extrudeSettings = { depth: thickness, // Ketebalan extrude bevelEnabled: false // Matikan sudut tumpul agar tembok rata }; const geometry = new THREE.ExtrudeGeometry(wallShape, extrudeSettings); const material = new THREE.MeshLambertMaterial({ color: 0xe0e0e0 }); const wallMesh = new THREE.Mesh(geometry, material); scene.add(wallMesh);
